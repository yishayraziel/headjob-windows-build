cmake_minimum_required(VERSION 3.22)
project(PluginProbe VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_PATH "" CACHE PATH "JUCE source folder")
if(JUCE_PATH)
 add_subdirectory("${JUCE_PATH}" JUCE)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_console_app(PluginProbe PRODUCT_NAME "PluginProbe")
target_sources(PluginProbe PRIVATE PluginProbe.cpp)
target_compile_definitions(PluginProbe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=1)
target_link_libraries(PluginProbe PRIVATE juce::juce_audio_utils)
