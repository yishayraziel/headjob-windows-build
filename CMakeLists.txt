cmake_minimum_required(VERSION 3.22)
project(HEADJOB VERSION 0.3.0)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(JUCE_PATH "" CACHE PATH "JUCE source folder")
if(JUCE_PATH)
 add_subdirectory("${JUCE_PATH}" JUCE)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
set(HEADJOB_FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND HEADJOB_FORMATS AU)
endif()
juce_add_plugin(HEADJOB COMPANY_NAME "HEADJOB Audio" BUNDLE_ID "audio.headjob.plugin" PLUGIN_MANUFACTURER_CODE Hjob PLUGIN_CODE HdJb FORMATS ${HEADJOB_FORMATS} PRODUCT_NAME "HEADJOB" COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(HEADJOBAssets SOURCES Assets/HEADJOB-Panel.png)
target_sources(HEADJOB PRIVATE Source/Plugin.cpp)
target_compile_definitions(HEADJOB PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(HEADJOB PRIVATE HEADJOBAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(HEADJOBTests PRODUCT_NAME "HEADJOBTests")
target_sources(HEADJOBTests PRIVATE Source/Tests.cpp)
target_compile_definitions(HEADJOBTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(HEADJOBTests PRIVATE HEADJOBAssets juce::juce_audio_utils juce::juce_dsp)
enable_testing()
add_test(NAME DSP COMMAND HEADJOBTests)
